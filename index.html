// কাজের ডামি ডেটা (নিশ্চিত করার জন্য যে ডেটা আছে) const jobs = [ { id: 1, title: "ইউটিউব ভিডিও দেখতে হবে এবং সাবস্ক্রাইব করতে হবে", reward: 5.00, employer: "Admin" }, { id: 2, title: "ফেসবুক পেজে লাইক ও শেয়ার করুন", reward: 3.50, employer: "Tanvir" }, { id: 3, title: "একটি ওয়েবসাইটে সাইন আপ করুন (রেফার ছাড়া)", reward: 15.00, employer: "Rahim" } ]; let userBalance = 0.00; // মূল ফাংশন যা কাজগুলো লোড করবে function loadJobs() { const jobListContainer = document.getElementById("job-list"); // নিশ্চিত হোন যে কন্টেইনারটি HTML এ আছে if (!jobListContainer) { console.error("ত্রুটি: 'job-list' আইডি যুক্ত ডিভটি HTML এ খুঁজে পাওয়া যায়নি।"); return; } jobListContainer.innerHTML = ""; // পুরানো তালিকা পরিষ্কার করুন jobs.forEach(job => { const jobCard = document.createElement("div"); jobCard.classList.add("job-card"); jobCard.innerHTML = `

${job.title}

পোস্টকারী: ${job.employer}

${job.reward.toFixed(2)} ৳
`; jobListContainer.appendChild(jobCard); }); } // ব্যালেন্স আপডেট করার ফাংশন function completeJob(reward) { userBalance += reward; const balanceElement = document.getElementById("balance"); if(balanceElement) { balanceElement.innerText = userBalance.toFixed(2); } alert(`অভিনন্দন! আপনি সফলভাবে কাজটি সম্পন্ন করেছেন এবং ${reward} টাকা অর্জন করেছেন।`); } // পেজ লোড হলে কাজগুলো দেখাবে - এটি একটি নিশ্চিত পদ্ধতি document.addEventListener("DOMContentLoaded", loadJobs);
